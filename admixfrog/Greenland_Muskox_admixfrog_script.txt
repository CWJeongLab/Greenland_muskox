#!/bin/bash

fn1=$1
of1=$2
lst1=$3   # modern sample
listf=$4  # outgroup takin sample
mask=$5
ref=$6

chr=$SLURM_ARRAY_TASK_ID

tn1=${of1}"_temp"

## extract modern samples & masked
strs=""; snum=0; while read iid; do if [[ "$snum" == 0 ]]; then strs=${iid}; else strs+=","${iid}; fi ; snum+=1; done < <(cut -f1 ${lst1}) # sample

bcftools view -Oz -s ${strs} ${fn1}.${chr}.vcf.gz -R ${mask} -o ${tn1}.${chr}.vcf.gz
bcftools index -t ${tn1}.${chr}.vcf.gz

## genotyping takin genomes using bcftools mpileup & call
vcfstr="${tn1}.${chr}.vcf.gz"
while read iid ibam; do
    bcftools mpileup -q 30 -Q 30 -C 50 -R ${tn1}.${chr}.vcf.gz -f ${ref} ${ibam} -Ou | bcftools call -c -V indels -Oz -o ./vcf/temp_${iid}.${chr}.vcf.gz
    bcftools index -t ./vcf/temp_${iid}.${chr}.vcf.gz
    vcfstr+=" ./vcf/temp_${iid}.${chr}.vcf.gz"
done < <(cat ${listf} | awk '{print $1, $3}')

bcftools merge ${vcfstr} -Oz -o ${of1}.${chr}.vcf.gz
bcftools index -t ${of1}.${chr}.vcf.gz

##### save the following script as "estimate_admixture_date.R" ####

#!/usr/bin/env Rscript

args <- commandArgs(trailingOnly = TRUE)
if (length(args) < 4) {
  stop("Usage: Rscript estimate_admixture_time.R <segment_file> <alpha> <min_length_cM> <sample_id>")
}

# Parse arguments
segment_file <- args[1]
alpha <- as.numeric(args[2])
min_cM <- as.numeric(args[3])
sample_id <- args[4]
of1 <- args[5]

# Load data
lengths_cM <- scan(segment_file, what = numeric(), quiet = TRUE)
lengths_cM <- lengths_cM[lengths_cM >= min_cM]
lengths_M <- lengths_cM / 100

if (length(lengths_M) < 10) {
  stop("Too few segments after filtering")
}

# Estimate admixture time
mean_len <- mean(lengths_M)
lambda_hat <- 1 / (mean_len - min_cM/100)
t_hat <- lambda_hat / (1-alpha)

# Save table
out_table <- data.frame(
  sample_id = sample_id,
  input_file = basename(segment_file),
  estimated_generations = round(t_hat, 2),
  mean_length_cM = mean(lengths_cM),
  alpha = alpha,
  min_cM = min_cM,
  n_segments = length(lengths_M)
)
write.table(out_table, paste0(of1, "_estimate.txt"), row.names = FALSE, quote = FALSE)

# Plot
pdf(paste0(of1, "_segment_lengths.pdf"), width = 8, height = 5)
hist(lengths_cM, breaks = 50, probability = TRUE,
     col = "skyblue", border = "gray30",
     main = paste0(sample_id, " - Estimated admixture time: ", round(t_hat, 1), " generations"),
     xlab = "Segment length (cM)", ylab = "Density")

x_vals <- seq(min(lengths_cM), max(lengths_cM), length.out = 50)
lambda <- 1 / ( (1-alpha) * (t_hat) )
pdf_vals <- (1/lambda)/100 * exp(- (1/lambda) * ( (x_vals -min_cM)/100))
lines(x_vals, pdf_vals, col = "red", lwd = 2)
#legend("topright", legend = paste0("λ = ", round(lambda, 2), ", α = ", alpha), col = "red", lwd = 2)
dev.off()

cat("[✓] Finished.\n")
cat("[✓] Estimated admixture time =", round(t_hat, 2), "generations\n")
cat("[✓] Output written to:", paste0(sample_id, "_estimate.csv"), "and plot\n")
