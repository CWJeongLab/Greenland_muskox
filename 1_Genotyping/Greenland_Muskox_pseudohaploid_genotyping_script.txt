## Save the following as "Run_pileup_pulldown_250804.sh"  ##

#!/bin/bash

listf=$1    ## A list of bam files (nr, iid, bamfile, pop, lid (if exists))
posf=$2     ## A position file to call genotype
snpf=$3     ## A snp file
refm=$4     ## A reference genome file
panel=$5    ## The number of SNP used in the panel
inum=$SLURM_ARRAY_TASK_ID

iid=($(awk -v inum="$inum" '{if (inum == $1) print $2}' ${listf}))              ## Individual ID
lid=($(awk -v inum="$inum" '{if (inum == $1 && NF == 5) print $5}' ${listf}))   ## Library ID (if exists)
bamf=($(awk -v inum="$inum" '{if (inum == $1) print $3}' ${listf}))             ## BAM file path

mkdir -p ./${iid}/; cd ./${iid}/

if [ -n "$lid" ]; then
    ## Make pileup file
    samtools mpileup -B -R -q30 -Q30 -l ${posf} -f ${refm} ${bamf} > ${lid}.pileup

    ## Run pileupCaller
    pileupCaller --randomHaploid --sampleNames ${lid} -f ${snpf} -e ${lid} < ${lid}.pileup

else

    ## Make pileup file
    samtools mpileup -B -R -q30 -Q30 -l ${posf} -f ${refm} ${bamf} > ${iid}.pileup

    ## Run pileupCaller
    pileupCaller --randomHaploid --sampleNames ${iid} -f ${snpf} -e ${iid} < ${iid}.pileup

fi

## Save the following as "Run_samtools_coverage_250805.sh"  ##

#!/bin/bash

listf=$1    ## A list of bam files (nr, iid, bamfile, pop, lid (if exists))
snpf=$2     ## A snp file
panel=$3    ## The number of SNP used in the panel
inum=$SLURM_ARRAY_TASK_ID

iid=($(awk -v inum="$inum" '{if (inum == $1) print $2}' ${listf}))              ## Individual ID
lid=($(awk -v inum="$inum" '{if (inum == $1 && NF == 5) print $5}' ${listf}))   ## Library ID (if exists)
pop=($(awk -v inum="$inum" '{if (inum == $1) print $4}' ${listf}))              ## Population
bamf=($(awk -v inum="$inum" '{if (inum == $1) print $3}' ${listf}))             ## BAM file path

mkdir -p ./${iid}/; cd ./${iid}/

if [ -n "$lid" ]; then
    ## Sex determination
    aC=$(samtools coverage -H -r 13 ${bamf} | awk '{print $7}')
    xC=$(samtools coverage -H -r X ${bamf} | awk '{print $7}')

    sex=$(awk -v x="$xC" -v a="$aC" '
        BEGIN {
            ratio = (a > 0) ? x/a : 0
            if (ratio > 0.7)      print "F"
            else if (ratio > 0.3) print "M"
            else                  print "U"
        }'
    )

    echo ${lid} ${sex} ${pop} | awk '{OFS="\t"} {print $1,$2,$3}' > ${lid}.ind

    ## Count the number and proportion of SNPs typed
    nsnps=($(wc -l ${snpf}))
    echo -e 'LID\tnCovered\tpCovered\taC\txC' > ${lid}.${panel}.cov.txt
    tnum1=($(awk '$1 != 9' ${lid}.geno | wc -l))
    echo ${lid} ${tnum1} ${nsnps} ${aC} ${xC} | awk '{OFS="\t"} {print $1,$2,$2/$3,$4,$5}' >> ${lid}.${panel}.cov.txt

else

    ## Sex determination
    aC=$(samtools coverage -H -r 13 ${bamf} | awk '{print $7}')
    xC=$(samtools coverage -H -r X ${bamf} | awk '{print $7}')

    sex=$(awk -v x="$xC" -v a="$aC" '
        BEGIN {
            ratio = (a > 0) ? x/a : 0
            if (ratio > 0.7)      print "F"
            else if (ratio > 0.3) print "M"
            else                  print "U"
        }'
    )

    echo ${iid} ${sex} ${pop} | awk '{OFS="\t"} {print $1,$2,$3}' > ${iid}.ind
    
    ## Count the number and proportion of SNPs typed
    nsnps=($(wc -l ${snpf}))
    echo -e 'IID\tnCovered\tpCovered\taC\txC' > ${iid}.${panel}.cov.txt
    tnum1=($(awk '$1 != 9' ${iid}.geno | wc -l))
    echo ${iid} ${tnum1} ${nsnps} ${aC} ${xC} | awk '{OFS="\t"} {print $1,$2,$2/$3,$4,$5}' >> ${iid}.${panel}.cov.txt
fi

## 2. Make mappability mask fasta file, apply the mappability filter and mac threshold  ##
##    Try Heng Li's mappability filtering (SNPable) for the human reference             ##
##    Follow the instruction from: http://lh3lh3.users.sourceforge.net/snpable.shtml    ##

## The scheme of the procedure is following behind.
## 1. split the reference genome to 35-bp non-overlapping chunks
## 2. map the chunks as short read to the reference 
## 3. score the mapping quality for read (whether it does be mapped to the same location or not). c=0~3. 35bp share same score.
## 4. we will use only regions of which score c=3, so we will make bed file for marking these regions. 

cd /home/References/Muskox/O.moschatus_RAN27_v1.0/

mkdir -p ./mappability/; cd ./mappability/

scf1=${pt1}"bwa_220818.sh"
of1="RAN27_v1.0_mask_35_50"  ## Mappability masked fasta file
tn1="temp1_"${of1}

## Split the reference genome to 35-bp non-overlapping chunks
splitfa ${refm} 35 | split -d -a 2 --numeric-suffixes=1 -l 20000000 \
    --filter='gzip > $FILE.gz' - ${tn1}_

idnmax=($(ls ${tn1}_*.gz | wc -l))
nperrun=9

## Run the job as a job array
sbatch --array=1-${idnmax}%${nperrun} -c 12 --mem 15000 -p amd --wrap=${scf1}" "${tn1}" "${refm}

## Crate raw mask file
## rewrite gen_raw_mask.pl because it generate one character for 35 bp. So I multiply 35 times.
gzip -dc ${tn1}_??.sam.gz | /opt/ohpc/pub/apps/seqability/seqbility-20091110/gen_raw_mask.pl > ${tn1}_1.fa 

## Create the final mask file with threshold 0.5 (-r 0.5)
gen_mask -l 35 -r 0.5 ${tn1}_1.fa > ${tn1}_2.fa

## Move the final file
mv ${tn1}_2.fa ${of1}.fa

## Save the following script as makeMappabilityMask.py ##

#!/usr/bin/env python

import gzip
import sys

INPUT = sys.argv[1]
OUTPUT= sys.argv[2]

class MaskGenerator:
  def __init__(self, filename, chr):
    self.lastCalledPos = -1
    self.lastStartPos = -1
    sys.stderr.write("making mask {}\n".format(filename))
    self.file = gzip.open(filename, "w")
    self.chr = chr

  # assume 1-based coordinate, output in bed format
  def addCalledPosition(self, pos):
    if self.lastCalledPos == -1:
      self.lastCalledPos = pos
      self.lastStartPos = pos
    elif pos == self.lastCalledPos + 1:
      self.lastCalledPos = pos
    else:
      self.file.write("{}\t{}\t{}\n".format(self.chr, self.lastStartPos - 1, self.lastCalledPos).encode())
      self.lastStartPos = pos
      self.lastCalledPos = pos

with open(INPUT, "r") as f:
  for line in f:
    if line.startswith('>'):
      chr = line.split()[0][1:]
      mask = MaskGenerator(OUTPUT.format(chr), chr)
      pos = 0
      continue
    for c in line.strip():
      pos += 1
      if pos % 1000000 == 0:
        sys.stderr.write("processing pos:{}\n".format(pos))
      if c == "3":
        mask.addCalledPosition(pos)
