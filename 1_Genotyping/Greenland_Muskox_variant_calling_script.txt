#!/bin/bash

listf=$1   ## A list of samples to be processed (cn, iid, bamdir, vcfdir, pop)
reff=$2    ## the reference genome list file
inum=$SLURM_ARRAY_TASK_ID

refn="Muskox"
ref=($(awk -v refn="$refn" '{if ($1 == refn) print $2}' ${reff}))

multigvcf="WGS.REF."${refn}".mapped.rmdup.q30."${inum}

GATKstr="java -Xmx12G -jar /opt/ohpc/pub/apps/gatk/gatk-4.1.2.0/gatk-package-4.1.2.0-local.jar"

## Create a string of input vcf for the combineGVCFs module
vcfstr=""
while read -r iid vcfdir; do
    invcf=($(ls ${vcfdir}/${iid}/${iid}.${refn}.mapped.rmdup.q30.${inum}.g.vcf.gz))
    vcfstr+=" -V "${invcf}
done < <(cat ${listf} | awk '{print $2, $4}')

mkdir -p ./251120_WGS_REF_EL012_Muskox/; cd ./251120_WGS_REF_EL012_Muskox/

## Run CombineGVCFs to merge all GVCFs into a single file
${GATKstr} CombineGVCFs -R ${ref} ${vcfstr} -O ${multigvcf}.g.vcf.gz

## Run GenotypeGVCFs to create variant calls
${GATKstr} GenotypeGVCFs -R ${ref} -V ${multigvcf}.g.vcf.gz -O ${multigvcf}.vcf.gz

#!/bin/bash

fn1=$1   ## input VCF prefix
of1=$2   ## output VCF prefix
refm=$3  ## Reference .fa

invcf=${fn1}"."${inum}".vcf.gz"
ovcf=${of1}"."${inum}

## Remove deletions and multi-allelic SNPs
## Take biSNPs with QUAL >= 30
gatk --java-options '-Xmx12G' SelectVariants -V ${invcf} -R ${refm} \
     --select "QUAL >= 30.0" --select-type-to-include SNP --restrict-alleles-to BIALLELIC \
     --exclude-filtered -O ${ovcf}.biSNP.vcf.gz



###############################################################################

## Save the following as "Variants_filtration_script_251124.sh"    ##

#!/bin/bash

fn1=$1   ## input VCF prefix
of1=$2   ## output VCF prefix
of2=$3   ## filter passed output VCF prefix
refm=$4  ## Reference .fa
inum=$SLURM_ARRAY_TASK_ID

invcf=${fn1}"."${inum}".biSNP.vcf.gz"
ovcf1=${of1}"."${inum}".vcf.gz"
ovcf2=${of2}"."${inum}".vcf.gz"

## Filter variants according to thresholds
## Median DP is about 1192
gatk --java-options '-Xmx12G' VariantFiltration -V ${invcf} -R ${refm} \
     -filter "DP < 200.0 || DP > 6000.0 || QD < 2.0 || SOR > 3.0 || FS > 60.0 || MQ < 35.0 || MQRankSum < -12.5 || ReadPosRankSum < -8.0 || ExcessHet > 10.0" \
     --filter-name "LowQual" -O ${ovcf1}

## Keep the passed variants only
bcftools view -i "FILTER='PASS'" -O z -o ${ovcf2} ${ovcf1}
gatk IndexFeatureFile -F ${ovcf2}
